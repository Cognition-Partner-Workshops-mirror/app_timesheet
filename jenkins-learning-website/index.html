Introduction

What is Jenkins?

The most popular open-source automation server that powers CI/CD pipelines worldwide

Continuous Integration

Jenkins automatically builds and tests code every time a developer pushes changes, catching bugs early in the development cycle.

Continuous Delivery

Automates the entire software delivery pipeline from code commit to production deployment with reliability and repeatability.

Plugin Ecosystem

Over 1800+ plugins extending Jenkins to integrate with virtually every tool in the DevOps ecosystem.

Distributed Builds

Distribute work across multiple machines for faster builds, testing on different platforms, and efficient resource utilization.

Why Jenkins Exists

Before CI

Manual Integration Hell

Developers manually merged code, leading to "integration hell" - conflicts discovered days or weeks after coding.

2004

Hudson Created

Kohsuke Kawaguchi at Sun Microsystems created Hudson to automate builds after getting tired of breaking code.

2011

Jenkins Born

After Oracle acquired Sun, the community forked Hudson as Jenkins - an open-source automation server.

Today

Industry Standard

Jenkins powers CI/CD for millions of projects, from startups to Fortune 500 companies, orchestrating the entire software delivery lifecycle.

Real World

Production Jenkins Pipeline

A real Java Spring Boot Banking Application pipeline with Jenkins

Banking Application - Payment Service

A production-grade Spring Boot microservice handling UPI transactions, Customer APIs, and payment processing.

Java 17 Spring Boot 3 PostgreSQL Kafka Redis Docker Kubernetes
Jenkinsfile Declarative Pipeline 
// Production Jenkinsfile - Banking Payment Service
pipeline {
    agent {
        kubernetes {
            yaml '''
            apiVersion: v1
            kind: Pod
            spec:
              containers:
              - name: maven
                image: maven:3.9-eclipse-temurin-17
                command: ['sleep', 'infinity']
              - name: docker
                image: docker:24-dind
                securityContext:
                  privileged: true
            '''
        }
    }

    environment {
        REGISTRY = 'registry.bank.internal'
        APP_NAME = 'payment-service'
        SONAR_HOST = 'https://sonar.bank.internal'
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
                script {
                    env.GIT_COMMIT_SHORT = sh(
                        script: 'git rev-parse --short HEAD',
                        returnStdout: true
                    ).trim()
                }
            }
        }

        stage('Build') {
            steps {
                container('maven') {
                    sh 'mvn clean compile -DskipTests'
                }
            }
        }

        stage('Unit Tests') {
            steps {
                container('maven') {
                    sh 'mvn test'
                }
            }
            post {
                always {
                    junit '**/target/surefire-reports/*.xml'
                    jacoco execPattern: '**/target/jacoco.exec'
                }
            }
        }

        stage('SonarQube Analysis') {
            steps {
                container('maven') {
                    withSonarQubeEnv('SonarQube') {
                        sh '''mvn sonar:sonar \
                            -Dsonar.projectKey=${APP_NAME} \
                            -Dsonar.host.url=${SONAR_HOST}'''
                    }
                }
            }
        }

        stage('Quality Gate') {
            steps {
                timeout(time: 5, unit: 'MINUTES') {
                    waitForQualityGate abortPipeline: true
                }
            }
        }

        stage('Package') {
            steps {
                container('maven') {
                    sh 'mvn package -DskipTests'
                    archiveArtifacts artifacts: 'target/*.jar'
                    fingerprint 'target/*.jar'
                }
            }
        }

        stage('Docker Build & Push') {
            steps {
                container('docker') {
                    script {
                        def image = docker.build(
                            "${REGISTRY}/${APP_NAME}:${GIT_COMMIT_SHORT}"
                        )
                        docker.withRegistry("https://${REGISTRY}",
                            'registry-credentials') {
                            image.push()
                            image.push('latest')
                        }
                    }
                }
            }
        }

        stage('Deploy to Staging') {
            steps {
                kubernetesDeploy(
                    configs: 'k8s/staging/',
                    kubeconfigId: 'k8s-staging-config'
                )
                sh '''kubectl rollout status \
                    deployment/${APP_NAME} \
                    -n payments-staging \
                    --timeout=300s'''
            }
        }

        stage('Integration Tests') {
            steps {
                sh '''./gradlew integrationTest \
                    -Dbase.url=http://${APP_NAME}.payments-staging.svc'''
            }
        }

        stage('Deploy to Production') {
            when {
                branch 'main'
            }
            input {
                message 'Deploy to production?'
                ok 'Deploy'
            }
            steps {
                kubernetesDeploy(
                    configs: 'k8s/production/',
                    kubeconfigId: 'k8s-prod-config'
                )
                sh '''kubectl rollout status \
                    deployment/${APP_NAME} \
                    -n payments-prod \
                    --timeout=300s'''
            }
        }

        stage('Health Check') {
            steps {
                script {
                    def response = httpRequest(
                        url: "http://${APP_NAME}.payments-prod.svc/actuator/health",
                        validResponseCodes: '200'
                    )
                    echo "Health: ${response.content}"
                }
            }
        }
    }

    post {
        success {
            slackSend(
                channel: '#deployments',
                color: 'good',
                message: "SUCCESS: ${APP_NAME} v${GIT_COMMIT_SHORT} deployed to production"
            )
        }
        failure {
            slackSend(
                channel: '#deployments',
                color: 'danger',
                message: "FAILED: ${APP_NAME} pipeline failed at ${currentBuild.currentResult}"
            )
        }
        always {
            cleanWs()
        }
    }
}
